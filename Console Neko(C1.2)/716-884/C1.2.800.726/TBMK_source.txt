[boot.asm]
============================================================

org 0x7C00
bits 16

start:
    jmp short real_start
    nop

; ============ FAT16 BPB 占位符 ============
bpb_oem:            db "T.B.M.K. "
bpb_bytes_per_sec:  dw 512
bpb_sec_per_cluster: db 8
bpb_reserved:       dw 1
bpb_num_fats:       db 2
bpb_root_entries:   dw 512
bpb_total_sec16:    dw 0
bpb_media:          db 0xF8
bpb_sec_per_fat:    dw 16
bpb_sec_per_track:  dw 63
bpb_num_heads:      dw 16
bpb_hidden:         dd 0
bpb_total_sec32:    dd 0
bpb_drive_num:      db 0x80
bpb_reserved2:      db 0
bpb_ext_sig:        db 0x29
bpb_vol_id:         dd 0x12345678
bpb_vol_label:      db "TBMK_DISK  "
bpb_fs_type:        db "FAT16   "

real_start:
    cli
    xor ax, ax
    mov ds, ax
    mov es, ax
    mov ss, ax
    mov sp, 0x7B00

    mov [boot_drive], dl
    mov byte [0x7F60], dl

    in al, 0x92
    or al, 2
    out 0x92, al

    mov ah, 0x00
    mov dl, [boot_drive]
    int 0x13

    ; CPU检测 (不变)
    mov eax, 0
    cpuid
    mov [0x7F00], ebx
    mov [0x7F04], edx
    mov [0x7F08], ecx
    mov byte [0x7F0C], 0

    mov eax, 0x80000000
    cpuid
    cmp eax, 0x80000004
    jb no_cpu_name

    mov eax, 0x80000002
    cpuid
    mov [0x7F10], eax
    mov [0x7F14], ebx
    mov [0x7F18], ecx
    mov [0x7F1C], edx

    mov eax, 0x80000003
    cpuid
    mov [0x7F20], eax
    mov [0x7F24], ebx
    mov [0x7F28], ecx
    mov [0x7F2C], edx

    mov eax, 0x80000004
    cpuid
    mov [0x7F30], eax
    mov [0x7F34], ebx
    mov [0x7F38], ecx
    mov [0x7F3C], edx

    mov byte [0x7F40], 0
    jmp cpu_done

no_cpu_name:
    mov byte [0x7F10], 0
cpu_done:

    mov dword [0x7F70], 0xB8000

    ; 根据启动驱动器选择 CHS 参数
    cmp byte [boot_drive], 0x80
    je hdd_boot

    ; ===== 软盘启动：LBA 1 = CHS (0/0/2) =====
    mov ax, 0x0201        ; AH=02(读), AL=01(先读1个扇区获取几何参数)
    mov ch, 0
    mov cl, 2
    mov dh, 0
    mov dl, [boot_drive]
    mov bx, 0x1000
    int 0x13
    jc disk_error

    ; 继续读剩余 49 个扇区
    mov ah, 0x02
    mov al, 49
    mov ch, 0
    mov cl, 3           ; 从扇区 3 继续
    mov dh, 0
    mov dl, [boot_drive]
    mov bx, 0x1200      ; 0x1000 + 512
    int 0x13
    jc disk_error
    jmp load_done

hdd_boot:
    ; ===== 硬盘启动：LBA 65 = 尝试用 CHS 读 =====
    ; QEMU 硬盘几何: 63 扇区/道, 16 磁头
    ; LBA 65 = 65/63 = 1 磁道余 2 → 柱面 0, 磁头 1, 扇区 3
    mov ax, 0x0201        ; 先读 1 个扇区测试
    mov ch, 0              ; 柱面 0
    mov cl, 3              ; 扇区 3
    mov dh, 1              ; 磁头 1
    mov dl, 0x80
    mov bx, 0x1000
    int 0x13
    jc try_hdd_chs2

    ; 继续读剩余 49 个扇区
    mov ah, 0x02
    mov al, 49
    mov ch, 0
    mov cl, 4
    mov dh, 1
    mov dl, 0x80
    mov bx, 0x1200
    int 0x13
    jc disk_error
    jmp load_done

try_hdd_chs2:
    ; 尝试柱面 1, 磁头 0, 扇区 2 (另一种 CHS 映射)
    mov ax, 0x0201
    mov ch, 1
    mov cl, 2
    mov dh, 0
    mov dl, 0x80
    mov bx, 0x1000
    int 0x13
    jc disk_error

    mov ah, 0x02
    mov al, 49
    mov ch, 1
    mov cl, 3
    mov dh, 0
    mov dl, 0x80
    mov bx, 0x1200
    int 0x13
    jc disk_error

load_done:
    cli
    lgdt [gdt_ptr]
    mov eax, cr0
    or eax, 1
    mov cr0, eax
    jmp 0x08:0x1000

disk_error:
    mov si, err_msg
    mov ah, 0x0E
print_loop:
    lodsb
    test al, al
    jz hang
    int 0x10
    jmp print_loop
hang:
    cli
halt_loop:
    hlt
    jmp halt_loop

boot_drive db 0
err_msg db 'Disk Error!', 0

align 4
gdt_start:
    dd 0, 0
    dd 0x0000FFFF, 0x00CF9A00
    dd 0x0000FFFF, 0x00CF9200
gdt_end:
gdt_ptr:
    dw gdt_end - gdt_start - 1
    dd gdt_start

times 510 - ($ - $$) db 0
dw 0xAA55

#include "keyboard.h"
#include "shell.h"
#include "version.h"
#include "except.h"
#include "syscall.h"
#include "disk.h"

#define VGA_BUFFER 0xB8000
#define COLOR 0x1F
#define BAR_COLOR 0x0F
#define VGA_CRTC_ADDR 0x3D4
#define VGA_CRTC_DATA 0x3D5

#define PAGE_SIZE 4096
#define MAX_PAGES 4096

#define KERNEL_START_LBA 65

void kernel_main(void) __attribute__((noreturn));

static unsigned char mem_bitmap[MAX_PAGES / 8];
unsigned int allocated_pages = 0;
volatile unsigned int timer_ticks = 0;

int cursor_x = 40;
int cursor_y = 10;
unsigned short cursor_backup = 0x1F00 | ' ';

static unsigned long long idt[256] __attribute__((aligned(8)));

static void set_idt_entry(int num, void* handler)
{
    unsigned long long addr = (unsigned long long)handler;
    unsigned short low = addr & 0xFFFF;
    unsigned short high = (addr >> 16) & 0xFFFF;

    idt[num] = ((unsigned long long)low)
             | (0x08ULL << 16)
             | (0x8EULL << 40)
             | ((unsigned long long)high << 48);
}

static void load_idt(void)
{
    struct {
        unsigned short limit;
        unsigned int base;
    } __attribute__((packed)) idt_ptr;

    idt_ptr.limit = sizeof(idt) - 1;
    idt_ptr.base = (unsigned int)&idt;

    __asm__ volatile("lidt %0" : : "m"(idt_ptr));
}

static void mm_init(void)
{
    unsigned int esp_val;
    __asm__ volatile("mov %%esp, %0" : "=r"(esp_val));

    unsigned int kernel_end_estimate = 0x90000 - esp_val + 0x20000;
    unsigned int kernel_pages = kernel_end_estimate / PAGE_SIZE + 1;
    if (kernel_pages < 1) kernel_pages = 1;
    if (kernel_pages > MAX_PAGES) kernel_pages = MAX_PAGES;

    for (int i = 0; i < MAX_PAGES / 8; i++) {
        mem_bitmap[i] = 0;
    }
    for (unsigned int i = 0; i < kernel_pages; i++) {
        mem_bitmap[i / 8] |= (1 << (i % 8));
    }
    allocated_pages = kernel_pages;
}

static void cursor_draw(void)
{
    volatile unsigned short *vga_ptr = (volatile unsigned short *)VGA_BUFFER;
    int pos = cursor_y * 80 + cursor_x;
    cursor_backup = vga_ptr[pos];
    unsigned short ch = vga_ptr[pos] & 0x00FF;
    vga_ptr[pos] = 0x7000 | ch;
}

static inline void outb(unsigned short port, unsigned char value)
{
    __asm__ volatile("outb %0, %1" : : "a"(value), "Nd"(port));
}

static inline unsigned char inb(unsigned short port)
{
    unsigned char value;
    __asm__ volatile("inb %1, %0" : "=a"(value) : "Nd"(port));
    return value;
}

static void pit_init(unsigned int hz)
{
    unsigned int divisor = 1193180 / hz;
    outb(0x43, 0x36);
    outb(0x40, divisor & 0xFF);
    outb(0x40, divisor >> 8);
}

__attribute__((naked)) void pit_handler(void)
{
    __asm__ volatile(
        "pushal\n"
        "incl _timer_ticks\n"
        "movb $0x20, %al\n"
        "outb %al, $0x20\n"
        "popal\n"
        "iret\n"
    );
}

static void disable_cursor(void)
{
    outb(VGA_CRTC_ADDR, 0x0A);
    outb(VGA_CRTC_DATA, 0x20);
}

static void beep(void)
{
    outb(0x43, 0xB6);
    outb(0x42, 0xA9);
    outb(0x42, 0x04);

    unsigned char tmp = inb(0x61);
    outb(0x61, tmp | 0x03);

    for (volatile int i = 0; i < 500000; i++) { }

    outb(0x61, tmp & 0xFC);
}

static void print_string(volatile unsigned short *vga, int row, const char *str)
{
    int pos = row * 80 + 2;
    while (*str) {
        if ((pos % 80) >= 78) break;
        vga[pos++] = (COLOR << 8) | *str++;
    }
}

static void draw_box(volatile unsigned short *vga)
{
    vga[80 + 0] = (COLOR << 8) | 0xC9;
    for (int i = 1; i < 79; i++) {
        vga[80 + i] = (COLOR << 8) | 0xCD;
    }
    vga[80 + 79] = (COLOR << 8) | 0xBB;

    for (int y = 2; y < 23; y++) {
        vga[y * 80 + 0] = (COLOR << 8) | 0xBA;
        vga[y * 80 + 79] = (COLOR << 8) | 0xBA;
    }

    vga[23 * 80 + 0] = (COLOR << 8) | 0xC8;
    for (int i = 1; i < 79; i++) {
        vga[23 * 80 + i] = (COLOR << 8) | 0xCD;
    }
    vga[23 * 80 + 79] = (COLOR << 8) | 0xBC;
}

void kernel_main(void)
{
    volatile unsigned short *vga = (volatile unsigned short *)VGA_BUFFER;

    disable_cursor();
    beep();
    mm_init();

    unsigned char boot_drive = *(unsigned char*)0x7F60;

    if (boot_drive == 0x00) {
        for (int i = 0; i < 80 * 25; i++)
            vga[i] = (COLOR << 8) | ' ';

        draw_box(vga);

        print_string(vga, 5, "  T.B.M.K. " KERNEL_VERSION " Installer");
        print_string(vga, 6, "  ==============================");
        print_string(vga, 8, "  [1] Install to hard disk");
        print_string(vga, 9, "  [2] Boot from hard disk");
        print_string(vga, 10, "  [3] Try live system");
        print_string(vga, 12, "  Press 1, 2 or 3 to select...");
        print_string(vga, 14, "  Ctrl+Alt+Del to reboot at any time.");

        unsigned char ins_ctrl = 0;
        unsigned char ins_alt = 0;
        int mode = 0;

        while (mode == 0) {
            if (inb(0x64) & 0x01) {
                unsigned char sc = inb(0x60);

                if (sc == 0x1D) ins_ctrl = 1;
                if (sc == 0x9D) ins_ctrl = 0;
                if (sc == 0x38) ins_alt = 1;
                if (sc == 0xB8) ins_alt = 0;

                if (sc == 0x53 && ins_ctrl && ins_alt && !(sc & 0x80)) {
                    while (inb(0x64) & 0x02);
                    outb(0x64, 0xFE);
                }

                if (sc == 0x02) mode = 1;
                if (sc == 0x03) mode = 2;
                if (sc == 0x04) mode = 3;
            }
        }

        if (mode == 1) {
            if (!disk_detect()) {
                print_string(vga, 16, "  No hard disk detected!");
                print_string(vga, 17, "  Install aborted.");
                print_string(vga, 18, "  Press Ctrl+Alt+Del to reboot.");

                while (1) {
                    if (inb(0x64) & 0x01) {
                        unsigned char sc2 = inb(0x60);
                        if (sc2 == 0x1D) ins_ctrl = 1;
                        if (sc2 == 0x9D) ins_ctrl = 0;
                        if (sc2 == 0x38) ins_alt = 1;
                        if (sc2 == 0xB8) ins_alt = 0;
                        if (sc2 == 0x53 && ins_ctrl && ins_alt && !(sc2 & 0x80)) {
                            while (inb(0x64) & 0x02);
                            outb(0x64, 0xFE);
                        }
                    }
                }
            }

            print_string(vga, 16, "  Installing...");

            // 先写内核到数据区（扇区 65 开始）
            unsigned char* kernel = (unsigned char*)0x1000;
            int write_failed = 0;
            for (int sec = 0; sec < 50; sec++) {
                if (disk_write(KERNEL_START_LBA + sec, kernel + sec * 512) != 0) {
                    write_failed = 1;
                    break;
                }
            }

            if (write_failed) {
                print_string(vga, 18, "  Write kernel failed!");
                while (1) {
                    if (inb(0x64) & 0x01) {
                        unsigned char sc2 = inb(0x60);
                        if (sc2 == 0x1D) ins_ctrl = 1;
                        if (sc2 == 0x9D) ins_ctrl = 0;
                        if (sc2 == 0x38) ins_alt = 1;
                        if (sc2 == 0xB8) ins_alt = 0;
                        if (sc2 == 0x53 && ins_ctrl && ins_alt && !(sc2 & 0x80)) {
                            while (inb(0x64) & 0x02);
                            outb(0x64, 0xFE);
                        }
                    }
                }
            }

            // 读取原始扇区 0（用于保留 BPB）
            unsigned char orig_sector[512];
            disk_read(0, orig_sector);

            // 准备新的引导扇区
            unsigned char buf[512];
            unsigned char* boot_sector = (unsigned char*)0x7C00;
            for (int i = 0; i < 512; i++) buf[i] = boot_sector[i];

            // 保留原始 BPB（偏移 3-61）
            if (orig_sector[11] == 0x00 && orig_sector[12] == 0x02) {
                for (int i = 3; i < 62; i++) {
                    buf[i] = orig_sector[i];
                }
            }

            // 在偏移 500 写入内核起始 LBA（小端）
            buf[500] = KERNEL_START_LBA & 0xFF;
            buf[501] = (KERNEL_START_LBA >> 8) & 0xFF;
            buf[502] = (KERNEL_START_LBA >> 16) & 0xFF;
            buf[503] = (KERNEL_START_LBA >> 24) & 0xFF;

            // 确保引导签名
            buf[510] = 0x55;
            buf[511] = 0xAA;

            // 写入引导扇区
            if (disk_write(0, buf) != 0) {
                print_string(vga, 18, "  Write boot sector failed!");
                while (1) {
                    if (inb(0x64) & 0x01) {
                        unsigned char sc2 = inb(0x60);
                        if (sc2 == 0x1D) ins_ctrl = 1;
                        if (sc2 == 0x9D) ins_ctrl = 0;
                        if (sc2 == 0x38) ins_alt = 1;
                        if (sc2 == 0xB8) ins_alt = 0;
                        if (sc2 == 0x53 && ins_ctrl && ins_alt && !(sc2 & 0x80)) {
                            while (inb(0x64) & 0x02);
                            outb(0x64, 0xFE);
                        }
                    }
                }
            }

            print_string(vga, 18, "  Installation complete!");
            print_string(vga, 19, "  Remove floppy and reboot.");
            print_string(vga, 20, "  Ctrl+Alt+Del to reboot.");

            while (1) {
                if (inb(0x64) & 0x01) {
                    unsigned char sc2 = inb(0x60);
                    if (sc2 == 0x1D) ins_ctrl = 1;
                    if (sc2 == 0x9D) ins_ctrl = 0;
                    if (sc2 == 0x38) ins_alt = 1;
                    if (sc2 == 0xB8) ins_alt = 0;
                    if (sc2 == 0x53 && ins_ctrl && ins_alt && !(sc2 & 0x80)) {
                        while (inb(0x64) & 0x02);
                        outb(0x64, 0xFE);
                    }
                }
            }
        }

        if (mode == 2) {
            print_string(vga, 16, "  Remove floppy disk, then");
            print_string(vga, 17, "  press Ctrl+Alt+Del to reboot.");
            print_string(vga, 18, "  System will boot from hard disk.");

            while (1) {
                if (inb(0x64) & 0x01) {
                    unsigned char sc2 = inb(0x60);
                    if (sc2 == 0x1D) ins_ctrl = 1;
                    if (sc2 == 0x9D) ins_ctrl = 0;
                    if (sc2 == 0x38) ins_alt = 1;
                    if (sc2 == 0xB8) ins_alt = 0;
                    if (sc2 == 0x53 && ins_ctrl && ins_alt && !(sc2 & 0x80)) {
                        while (inb(0x64) & 0x02);
                        outb(0x64, 0xFE);
                    }
                }
            }
        }
    }

    set_idt_entry(0, except_handler_0);
    set_idt_entry(6, except_handler_6);
    set_idt_entry(13, except_handler_13);
    set_idt_entry(14, except_handler_14);
    set_idt_entry(0x20, pit_handler);
    set_idt_entry(0x80, syscall_handler);
    set_idt_entry(0x21, keyboard_handler);
    load_idt();

    keyboard_init();
    pit_init(100);
    outb(0x21, inb(0x21) & 0xFE);

    for (int i = 0; i < 80 * 25; i++)
        vga[i] = (COLOR << 8) | ' ';

    for (int i = 0; i < 80; i++) {
        vga[i] = 0x7000 | ' ';
    }
    char *title = " T.B.M.K. Micro Kernel - Main Page ";
    int title_len = 0;
    while (title[title_len]) title_len++;
    int title_start = (80 - title_len) / 2;
    for (int i = 0; i < title_len; i++) {
        vga[title_start + i] = 0x7000 | title[i];
    }

    print_string(vga, 2, KERNEL_FULL_STRING);
    print_string(vga, 4, "  _______   ______   .___  ___.  __  ___  ");
    print_string(vga, 5, " |       | |   _  |  |   ||   | |  |/  /  ");
    print_string(vga, 6, " |_     _| |  |_) |  |   ||   | |  '  /   ");
    print_string(vga, 7, "   |   |   |   _ <   |   ||   | |    <    ");
    print_string(vga, 8, "   |   |   |  |_) |  |   ||   | |  .  \\  ");
    print_string(vga, 9, "   |___|   |______/  |___||___| |__|\\__\\ ");
    print_string(vga, 11, "   Tiny Bare-Metal Micro Kernel");
    print_string(vga, 21, "  [!][Arrow keys] Move cursor");
    print_string(vga, 22, "  [!][Ctrl+S] Shell | [Ctrl+Alt+Del] Reboot");

    draw_box(vga);

    shell_init();
    cursor_draw();

    while (1) {
        __asm__ volatile("hlt");
    }
}

    int pos = row * 80 + col;
    for (int i = 0; i < 30; i++) {

    }
}

__attribute__((naked)) void keyboard_handler(void)
{
    __asm__ volatile(
        "pushal\n"
        "cld\n"
        "call _keyboard_callback\n"
        "popal\n"
        "iret\n"
    );
}

void keyboard_callback(void)
{
    unsigned char scancode = inb(0x60);
    static unsigned char ctrl_pressed = 0;
    static unsigned char alt_pressed = 0;
    static unsigned char ext_key = 0;
    static char key_status[40];

    if (scancode == 0xE0) {
        ext_key = 1;
        outb(0x20, 0x20);
        return;
    }

    const char* name = get_key_name_raw(scancode);
    if (scancode & 0x80) {
        int i = 0;
        while (name[i] && i < 20) { key_status[i] = name[i]; i++; }
        const char* rel = " Released";
        int j = 0;
        while (rel[j] && (i + j) < 39) { key_status[i + j] = rel[j]; j++; }
        key_status[i + j] = '\0';
    } else {
        int i = 0;
        while (name[i] && i < 39) { key_status[i] = name[i]; i++; }
        key_status[i] = '\0';
    }

    int len = 0;
    while (key_status[len]) len++;

    show_string("Key Status:[", 24, 0);
    show_string(key_status, 24, 13);
    show_string("]", 24, 13 + len + 1);

    if (scancode == 0x1D) {
        ctrl_pressed = 1;
    } else if (scancode == 0x9D) {
        ctrl_pressed = 0;
    }

    if (scancode == 0x38) {
        alt_pressed = 1;
    } else if (scancode == 0xB8) {
        alt_pressed = 0;
    }

    if (scancode == 0x53 && ctrl_pressed && alt_pressed && !(scancode & 0x80)) {
        reboot();
    }

    if (shell_is_open()) {
        if (scancode == 0x01 && !(scancode & 0x80)) {
            shell_close();
        } else if (scancode == 0x26 && ctrl_pressed && !(scancode & 0x80)) {
            shell_clear();
        } else {
            shell_input(scancode);
        }
        outb(0x20, 0x20);
        return;
    }

    if (ext_key && !(scancode & 0x80)) {
        volatile unsigned short *vga = (volatile unsigned short *)VGA_BUFFER;
        int old_pos = cursor_y * 80 + cursor_x;
        vga[old_pos] = cursor_backup;

        switch (scancode) {
            case 0x48: cursor_y--; break;
            case 0x50: cursor_y++; break;
            case 0x4B: cursor_x--; break;
            case 0x4D: cursor_x++; break;
        }
        if (cursor_x < 1) cursor_x = 1;
        if (cursor_x > 78) cursor_x = 78;
        if (cursor_y < 2) cursor_y = 2;
        if (cursor_y > 22) cursor_y = 22;

        int new_pos = cursor_y * 80 + cursor_x;
        cursor_backup = vga[new_pos];
        unsigned short ch = vga[new_pos] & 0x00FF;
        vga[new_pos] = 0x7000 | ch;
    }
    ext_key = 0;

    if (scancode == 0x1F && ctrl_pressed) {
        shell_show();
    }

    outb(0x20, 0x20);
}

void keyboard_init(void)
{
    outb(0x20, 0x11);
    outb(0xA0, 0x11);
    outb(0x21, 0x20);
    outb(0xA1, 0x28);
    outb(0x21, 0x04);
    outb(0xA1, 0x02);
    outb(0x21, 0x01);
    outb(0xA1, 0x01);
    outb(0x21, 0xFD);
    outb(0xA1, 0xFF);

    for (int i = 0; i < 10; i++) {
        if (inb(0x64) & 0x01) {
            inb(0x60);
        }
    }

    __asm__ volatile("sti");
}

static inline void outb(unsigned short port, unsigned char value)
{
    __asm__ volatile("outb %0, %1" : : "a"(value), "Nd"(port));
}

static inline unsigned char inb(unsigned short port)
{
    unsigned char value;
    __asm__ volatile("inb %1, %0" : "=a"(value) : "Nd"(port));
    return value;
}

static void put_char(int x, int y, unsigned char c, unsigned char attr)
{
    volatile unsigned short *vga = (volatile unsigned short *)VGA_BUFFER;
    if (x >= 1 && x < WIDTH - 1 && y >= 2 && y < 23) {
        vga[y * WIDTH + x] = (attr << 8) | c;
    }
}

static void draw_shell_box(volatile unsigned short *vga)
{
    vga[80 + 0] = (COLOR_INFO << 8) | 0xC9;
    for (int i = 1; i < 79; i++) {
        vga[80 + i] = (COLOR_INFO << 8) | 0xCD;
    }
    vga[80 + 79] = (COLOR_INFO << 8) | 0xBB;

    for (int y = 2; y < 23; y++) {
        vga[y * 80 + 0] = (COLOR_INFO << 8) | 0xBA;
        vga[y * 80 + 79] = (COLOR_INFO << 8) | 0xBA;
    }

    vga[23 * 80 + 0] = (COLOR_INFO << 8) | 0xC8;
    for (int i = 1; i < 79; i++) {
        vga[23 * 80 + i] = (COLOR_INFO << 8) | 0xCD;
    }
    vga[23 * 80 + 79] = (COLOR_INFO << 8) | 0xBC;
}

static void scroll_up(void)
{
    volatile unsigned short *vga = (volatile unsigned short *)VGA_BUFFER;
    for (int y = 2; y < 22; y++) {
        for (int x = 1; x < WIDTH - 1; x++) {
            vga[y * WIDTH + x] = vga[(y + 1) * WIDTH + x];
        }
    }
    for (int x = 1; x < WIDTH - 1; x++) {
        put_char(x, 22, ' ', COLOR_INFO);
    }
}

static int str_len(const char* s)
{
    int n = 0;
    while (s[n]) n++;
    return n;
}

static int str_cmp(const char* a, const char* b)
{
    while (*a && *b && *a == *b) { a++; b++; }
    return *a - *b;
}

static void shell_print_line_color(const char* str, unsigned char attr)
{
    if (cursor_y > 22) {
        scroll_up();
        cursor_y = 22;
    }

    int x = 1;
    while (*str && x < WIDTH - 1) {
        put_char(x, cursor_y, *str, attr);
        x++;
        str++;
    }
    cursor_y++;

    if (cursor_y > 22) {
        scroll_up();
        cursor_y = 22;
    }
}

static void shell_print_line(const char* str)
{
    shell_print_line_color(str, COLOR_INFO);
}

        while (*s2 && p < 70) buf[p++] = *s2++;
        buf[p++] = ' ';
        buf[p++] = '(';
        char num[10]; int np = 0;
        int n = bytes_read;
        if (n == 0) num[np++] = '0';
        while (n > 0) { num[np++] = '0' + (n % 10); n /= 10; }
        while (np > 0) buf[p++] = num[--np];
        buf[p++] = ' '; buf[p++] = 'b'; buf[p++] = 'y'; buf[p++] = 't'; buf[p++] = 'e'; buf[p++] = 's';
        buf[p++] = ')';
        buf[p] = '\0';
        shell_print_line_color(buf, COLOR_OK);

        void (*user_func)() = (void (*)())0x200000;
        user_func();
        shell_print_line("[!] Program returned.");

    } else {
        shell_print_line_color("[X] Unknown command. Type HELP for list.", COLOR_ERR);
    }
}

static char scancode_to_ascii(unsigned char scancode)
{
    int upper = shift_pressed ^ caps_lock;
    switch (scancode)
    {
        case 0x02: return upper ? '!' : '1';
        case 0x03: return upper ? '@' : '2';
        case 0x04: return upper ? '#' : '3';
        case 0x05: return upper ? '$' : '4';
        case 0x06: return upper ? '%' : '5';
        case 0x07: return upper ? '^' : '6';
        case 0x08: return upper ? '&' : '7';
        case 0x09: return upper ? '*' : '8';
        case 0x0A: return upper ? '(' : '9';
        case 0x0B: return upper ? ')' : '0';
        case 0x0C: return upper ? '_' : '-';
        case 0x0D: return upper ? '+' : '=';
        case 0x10: return upper ? 'Q' : 'q';
        case 0x11: return upper ? 'W' : 'w';
        case 0x12: return upper ? 'E' : 'e';
        case 0x13: return upper ? 'R' : 'r';
        case 0x14: return upper ? 'T' : 't';
        case 0x15: return upper ? 'Y' : 'y';
        case 0x16: return upper ? 'U' : 'u';
        case 0x17: return upper ? 'I' : 'i';
        case 0x18: return upper ? 'O' : 'o';
        case 0x19: return upper ? 'P' : 'p';
        case 0x1A: return upper ? '{' : '[';
        case 0x1B: return upper ? '}' : ']';
        case 0x1E: return upper ? 'A' : 'a';
        case 0x1F: return upper ? 'S' : 's';
        case 0x20: return upper ? 'D' : 'd';
        case 0x21: return upper ? 'F' : 'f';
        case 0x22: return upper ? 'G' : 'g';
        case 0x23: return upper ? 'H' : 'h';
        case 0x24: return upper ? 'J' : 'j';
        case 0x25: return upper ? 'K' : 'k';
        case 0x26: return upper ? 'L' : 'l';
        case 0x27: return upper ? ':' : ';';
        case 0x28: return upper ? '"' : '\'';
        case 0x29: return upper ? '~' : '`';
        case 0x2B: return upper ? '|' : '\\';
        case 0x2C: return upper ? 'Z' : 'z';
        case 0x2D: return upper ? 'X' : 'x';
        case 0x2E: return upper ? 'C' : 'c';
        case 0x2F: return upper ? 'V' : 'v';
        case 0x30: return upper ? 'B' : 'b';
        case 0x31: return upper ? 'N' : 'n';
        case 0x32: return upper ? 'M' : 'm';
        case 0x33: return upper ? '<' : ',';
        case 0x34: return upper ? '>' : '.';
        case 0x35: return upper ? '?' : '/';
        case 0x39: return ' ';
        default: return 0;
    }
}

static void show_prompt(void)
{
    if (cursor_y > 22) {
        scroll_up();
        cursor_y = 22;
    }
    put_char(1, cursor_y, '>', COLOR_INFO);
    put_char(2, cursor_y, ' ', COLOR_INFO);
    input_pos = 3;
    put_char(input_pos, cursor_y, '_', COLOR_INFO);
}

void shell_input(unsigned char scancode)
{
    if (!shell_open) return;
    if(scancode == 0x2A || scancode == 0x36) { shift_pressed = 1; return; }
    if(scancode == 0xAA || scancode == 0xB6) { shift_pressed = 0; return; }
    if(scancode == 0x3A) { caps_lock = !caps_lock; return; }
    if (scancode & 0x80) return;

    if (scancode == 0x48) {
        if (history_count > 0 && history_pos < 4) {
            if (history_pos == -1) history_pos = (history_count - 1) % 5;
            else history_pos = (history_pos - 1 + 5) % 5;
            for (int i = 3; i < WIDTH - 1; i++) put_char(i, cursor_y, ' ', COLOR_INFO);
            put_char(1, cursor_y, '>', COLOR_INFO);
            put_char(2, cursor_y, ' ', COLOR_INFO);
            int h = history_pos;
            for (int i = 0; history[h][i] && i < 75; i++) put_char(3 + i, cursor_y, history[h][i], COLOR_INFO);
            input_pos = 3;
            for (int i = 0; history[h][i] && i < 75; i++) { input_buf[i] = history[h][i]; input_pos++; }
            input_buf[input_pos - 3] = '\0';
            put_char(input_pos, cursor_y, '_', COLOR_INFO);
        }
        return;
    }
    if (scancode == 0x50) {
        if (history_pos != -1) {
            history_pos = (history_pos + 1) % 5;
            if (history_pos >= history_count || history_count == 0) {
                history_pos = -1;
                for (int i = 3; i < WIDTH - 1; i++) put_char(i, cursor_y, ' ', COLOR_INFO);
                put_char(1, cursor_y, '>', COLOR_INFO);
                put_char(2, cursor_y, ' ', COLOR_INFO);
                input_buf[0] = '\0'; input_pos = 3;
                put_char(input_pos, cursor_y, '_', COLOR_INFO);
            } else {
                for (int i = 3; i < WIDTH - 1; i++) put_char(i, cursor_y, ' ', COLOR_INFO);
                put_char(1, cursor_y, '>', COLOR_INFO);
                put_char(2, cursor_y, ' ', COLOR_INFO);
                int h = history_pos;
                for (int i = 0; history[h][i] && i < 75; i++) put_char(3 + i, cursor_y, history[h][i], COLOR_INFO);
                input_pos = 3;
                for (int i = 0; history[h][i] && i < 75; i++) { input_buf[i] = history[h][i]; input_pos++; }
                input_buf[input_pos - 3] = '\0';
                put_char(input_pos, cursor_y, '_', COLOR_INFO);
            }
        }
        return;
    }

    if (scancode == 0x0F) {
        if (input_pos <= 3) return;
        input_buf[input_pos - 3] = '\0';
        const char* cmds[] = {"abt", "cat", "cls", "cpu", "crash", "date", "dir", "echo",
                              "help", "load", "mem", "rbt", "run", "setup", "syscall",
                              "uptime", "ver"};
        int match_count = 0;
        int match_idx = 0;
        int cmd_count = 17;
        for (int i = 0; i < cmd_count; i++) {
            int j;
            for (j = 0; input_buf[j] && cmds[i][j]; j++) {
                if (input_buf[j] != cmds[i][j] &&
                    input_buf[j] != cmds[i][j] - 'a' + 'A') break;
            }
            if (input_buf[j] == '\0') { match_count++; match_idx = i; }
        }
        if (match_count == 1) {
            for (int i = 3; i < WIDTH - 1; i++) put_char(i, cursor_y, ' ', COLOR_INFO);
            put_char(1, cursor_y, '>', COLOR_INFO);
            put_char(2, cursor_y, ' ', COLOR_INFO);
            int i;
            for (i = 0; cmds[match_idx][i]; i++) {
                put_char(3 + i, cursor_y, cmds[match_idx][i], COLOR_INFO);
                input_buf[i] = cmds[match_idx][i];
            }
            input_buf[i] = '\0';
            input_pos = 3 + i;
            put_char(input_pos, cursor_y, '_', COLOR_INFO);
        } else if (match_count > 1) {
            cursor_y++;
            if (cursor_y > 22) { scroll_up(); cursor_y = 22; }
            for (int i = 0; i < cmd_count; i++) {
                int j;
                for (j = 0; input_buf[j] && cmds[i][j]; j++) {
                    if (input_buf[j] != cmds[i][j] &&
                        input_buf[j] != cmds[i][j] - 'a' + 'A') break;
                }
                if (input_buf[j] == '\0') {
                    put_char(1, cursor_y, cmds[i][0], COLOR_INFO);
                    for (int k = 1; cmds[i][k]; k++) put_char(1 + k, cursor_y, cmds[i][k], COLOR_INFO);
                    cursor_y++;
                    if (cursor_y > 22) { scroll_up(); cursor_y = 22; }
                }
            }
            show_prompt();
        }
        return;
    }

    if (scancode == 0x0E) {
        if (input_pos > 3) {
            input_pos--;
            put_char(input_pos, cursor_y, ' ', COLOR_INFO);
            put_char(input_pos + 1, cursor_y, ' ', COLOR_INFO);
            put_char(input_pos, cursor_y, '_', COLOR_INFO);
            input_buf[input_pos - 3] = '\0';
        }
        return;
    }

    if (scancode == 0x1C) {
        put_char(input_pos, cursor_y, ' ', COLOR_INFO);
        input_buf[input_pos - 3] = '\0';
        if (input_buf[0] != '\0') {
            int h = history_count % 5;
            for (int i = 0; i < 80; i++) history[h][i] = input_buf[i];
            history_count++;
            history_pos = -1;
        }
        cursor_y++;
        if (cursor_y > 22) { scroll_up(); cursor_y = 22; }
        input_pos = 3;
        execute_command(input_buf);
        input_buf[0] = '\0';
        input_pos = 3;
        show_prompt();
        return;
    }

    char c = scancode_to_ascii(scancode);
    if (c && input_pos < WIDTH - 3) {
        put_char(input_pos, cursor_y, c, COLOR_INFO);
        input_buf[input_pos - 3] = c;
        input_pos++;
        put_char(input_pos, cursor_y, '_', COLOR_INFO);
    }
}

    }
}

void sys_print(const char* str)
{
    volatile unsigned short *vga = (volatile unsigned short *)VGA_BUFFER;
    int row = 6;
    int col = 2;

    while (*str) {
        if (*str == '\n') { row++; col = 2; str++; continue; }
        if (row > 20) break;
        vga[row * 80 + col] = 0x1F00 | *str;
        col++; str++;
        if (col > 70) { col = 2; row++; }
    }
}

void sys_clear(void)
{
    volatile unsigned short *vga = (volatile unsigned short *)VGA_BUFFER;
    for (int i = 0; i < 80 * 25; i++)
        vga[i] = 0x1F00 | ' ';
}

void sys_reboot(void)
{
    while (inb(0x64) & 0x02);
    outb(0x64, 0xFE);
}

__attribute__((naked)) void sys_exec(void)
{
    __asm__ volatile(
        "movl $0x90000, %%esp\n"  // 重置栈
        "movl $0x200000, %%eax\n"
        "jmp *%%eax\n"
        :
        :
        : "eax"
    );
}

__attribute__((naked)) void sys_exit(void)
{
    __asm__ volatile(
        "call _sys_reboot\n"
    );
}

__attribute__((naked)) void syscall_handler(void)
{
    __asm__ volatile(
        "pushal\n"
        "cmpl $1, %%eax\n"
        "je 1f\n"
        "cmpl $2, %%eax\n"
        "je 2f\n"
        "cmpl $3, %%eax\n"
        "je 3f\n"
        "cmpl $4, %%eax\n"
        "je 4f\n"
        "cmpl $5, %%eax\n"
        "je 5f\n"
        "jmp 9f\n"
        "1:\n"
        "pushl %%ebx\n"
        "call _sys_print\n"
        "addl $4, %%esp\n"
        "jmp 9f\n"
        "2:\n"
        "call _sys_clear\n"
        "jmp 9f\n"
        "3:\n"
        "call _sys_reboot\n"
        "jmp 9f\n"
        "4:\n"
        "call _sys_exec\n"
        "jmp 9f\n"
        "5:\n"
        "call _sys_exit\n"
        "jmp 9f\n"
        "9:\n"
        "popal\n"
        "iret\n"
        :
        :
        : "eax", "ebx"
    );
}

static inline unsigned short inw(unsigned short port)
{
    unsigned short value;
    __asm__ volatile("inw %1, %0" : "=a"(value) : "Nd"(port));
    return value;
}

static inline void outw(unsigned short port, unsigned short value)
{
    __asm__ volatile("outw %0, %1" : : "a"(value), "Nd"(port));
}

int disk_detect(void)
{
    outb(0x1F6, 0xA0);
    outb(0x1F7, 0xEC);

    unsigned char status = inb(0x1F7);
    if (status == 0xFF || status == 0x00) return 0;

    int timeout = 100000;
    while ((inb(0x1F7) & 0x80) && timeout > 0) timeout--;
    if (timeout == 0) return 0;
    if (inb(0x1F7) & 0x01) return 0;

    for (int i = 0; i < 256; i++) inw(0x1F0);
    return 1;
}

int disk_read(unsigned int lba, unsigned char* buf)
{
    while (inb(0x1F7) & 0x80);
    outb(0x1F2, 1);
    outb(0x1F3, lba & 0xFF);
    outb(0x1F4, (lba >> 8) & 0xFF);
    outb(0x1F5, (lba >> 16) & 0xFF);
    outb(0x1F6, 0xE0 | ((lba >> 24) & 0x0F));
    outb(0x1F7, 0x20);
    while (!(inb(0x1F7) & 0x08));
    unsigned short* buf16 = (unsigned short*)buf;
    for (int i = 0; i < 256; i++) {
        buf16[i] = inw(0x1F0);
    }
    return 0;
}

int disk_write(unsigned int lba, const unsigned char* buf)
{
    while (inb(0x1F7) & 0x80);
    outb(0x1F2, 1);
    outb(0x1F3, lba & 0xFF);
    outb(0x1F4, (lba >> 8) & 0xFF);
    outb(0x1F5, (lba >> 16) & 0xFF);
    outb(0x1F6, 0xE0 | ((lba >> 24) & 0x0F));
    outb(0x1F7, 0x30);
    while (!(inb(0x1F7) & 0x08));
    unsigned short* buf16 = (unsigned short*)buf;
    for (int i = 0; i < 256; i++) {
        outw(0x1F0, buf16[i]);
    }
    outb(0x1F7, 0xE7);
    while (inb(0x1F7) & 0x80);
    return 0;
}

    kernel_end = .;

    /DISCARD/ : {
        *(.comment)
        *(.note*)
        *(.eh_frame)
    }
}

#include "fat.h"
#include "disk.h"

#define SECTOR_SIZE 512

fat_fs_t g_fs;
static file_handle_t g_handle;

static int str_casecmp(const char* a, const char* b, int n)
{
    for (int i = 0; i < n; i++) {
        char ca = a[i];
        char cb = b[i];
        if (ca >= 'a' && ca <= 'z') ca -= 32;
        if (cb >= 'a' && cb <= 'z') cb -= 32;
        if (ca != cb) return ca - cb;
        if (ca == 0) return 0;
    }
    return 0;
}

static int str_len(const char* s)
{
    int n = 0;
    while (s[n]) n++;
    return n;
}

static void str_copy(char* dst, const char* src, int max)
{
    int i;
    for (i = 0; i < max - 1 && src[i]; i++) {
        dst[i] = src[i];
    }
    dst[i] = '\0';
}

static int fat_read_fat(void)
{
    unsigned int fat_start = g_fs.reserved_sectors;
    int sectors_to_read = g_fs.sectors_per_fat;
    if (sectors_to_read > 128) sectors_to_read = 128;

    for (int i = 0; i < sectors_to_read; i++) {
        unsigned char buf[512];
        if (disk_read(fat_start + i, buf) != 0) return -1;
        unsigned short* cache_ptr = &g_fs.fat_cache[i * 256];
        unsigned short* buf_ptr = (unsigned short*)buf;
        for (int j = 0; j < 256; j++) {
            cache_ptr[j] = buf_ptr[j];
        }
    }
    g_fs.fat_loaded = 1;
    return 0;
}

static unsigned short fat_next_cluster(unsigned short cluster)
{
    if (cluster < 2 || cluster >= g_fs.total_clusters + 2) return 0xFFFF;
    unsigned short value = g_fs.fat_cache[cluster];
    if (value >= 0xFFF8) return 0xFFFF;
    if (value == 0xFFF7) return 0xFFF7;
    if (value == 0)     return 0xFFFF;
    return value;
}

static unsigned int cluster_to_sector(unsigned short cluster)
{
    return g_fs.first_data_sector + (cluster - 2) * g_fs.sectors_per_cluster;
}

static void fat_name_to_str(const unsigned char* raw_name, char* out)
{
    int p = 0;
    for (int i = 0; i < 8; i++) {
        if (raw_name[i] == ' ' || raw_name[i] == 0) break;
        out[p++] = raw_name[i];
    }
    if (raw_name[8] != ' ' && raw_name[8] != 0) {
        out[p++] = '.';
        for (int i = 8; i < 11; i++) {
            if (raw_name[i] == ' ' || raw_name[i] == 0) break;
            out[p++] = raw_name[i];
        }
    }
    out[p] = '\0';
}

static void str_to_fat_name(const char* input, unsigned char* name, unsigned char* ext)
{
    for (int i = 0; i < 8; i++) name[i] = ' ';
    for (int i = 0; i < 3; i++) ext[i] = ' ';
    int dot_pos = -1;
    int len = str_len(input);
    for (int i = 0; i < len; i++) {
        if (input[i] == '.') { dot_pos = i; break; }
    }
    int main_len = (dot_pos >= 0) ? dot_pos : len;
    if (main_len > 8) main_len = 8;
    for (int i = 0; i < main_len; i++) {
        char c = input[i];
        if (c >= 'a' && c <= 'z') c -= 32;
        name[i] = c;
    }
    if (dot_pos >= 0) {
        int ext_start = dot_pos + 1;
        int ext_len = len - ext_start;
        if (ext_len > 3) ext_len = 3;
        for (int i = 0; i < ext_len; i++) {
            char c = input[ext_start + i];
            if (c >= 'a' && c <= 'z') c -= 32;
            ext[i] = c;
        }
    }
}

int fat_init(void)
{
    unsigned char boot_sector[512];

    if (disk_read(0, boot_sector) != 0) return -1;

    bpb_t* bpb = 0;
    unsigned int partition_start = 0;

    bpb = (bpb_t*)&boot_sector[11];

    if (boot_sector[11] == 0x00 && boot_sector[12] == 0x02) {
        partition_start = 0;
    } else {
        unsigned int* part_start = (unsigned int*)&boot_sector[0x1C6];
        partition_start = *part_start;
        if (partition_start == 0 || partition_start > 100000) partition_start = 63;
        if (disk_read(partition_start, boot_sector) != 0) return -1;
        bpb = (bpb_t*)&boot_sector[11];
        if (boot_sector[11] != 0x00 || boot_sector[12] != 0x02) {
            partition_start = 2048;
            if (disk_read(partition_start, boot_sector) != 0) return -1;
            bpb = (bpb_t*)&boot_sector[11];
        }
    }

    if (bpb->bytes_per_sector != 512) return -1;

    g_fs.sectors_per_cluster = bpb->sectors_per_cluster;
    g_fs.reserved_sectors   = bpb->reserved_sectors + partition_start;
    g_fs.num_fats           = bpb->num_fats;
    g_fs.root_entries       = bpb->root_entries;
    g_fs.sectors_per_fat    = bpb->sectors_per_fat;
    g_fs.root_dir_sectors   = (g_fs.root_entries * 32 + 511) / 512;

    unsigned int total_sectors = bpb->total_sectors_16;
    if (total_sectors == 0) total_sectors = bpb->total_sectors_32;

    unsigned int data_sectors = total_sectors
                              - g_fs.reserved_sectors
                              - g_fs.num_fats * g_fs.sectors_per_fat
                              - g_fs.root_dir_sectors;

    g_fs.total_clusters = data_sectors / g_fs.sectors_per_cluster;
    g_fs.first_data_sector = g_fs.reserved_sectors
                           + g_fs.num_fats * g_fs.sectors_per_fat
                           + g_fs.root_dir_sectors;

    if (g_fs.sectors_per_cluster == 0 || g_fs.sectors_per_cluster > 128) return -1;
    if (g_fs.sectors_per_fat == 0 || g_fs.sectors_per_fat > 256) return -1;
    if (g_fs.root_entries == 0) return -1;

    g_fs.fat_loaded = 0;

    if (fat_read_fat() != 0) return -1;

    return 0;
}

int fat_list_root(void (*callback)(const file_info_t*))
{
    unsigned int root_start = g_fs.reserved_sectors + g_fs.num_fats * g_fs.sectors_per_fat;
    unsigned char buf[512];
    file_info_t info;

    for (unsigned int i = 0; i < g_fs.root_dir_sectors; i++) {
        if (disk_read(root_start + i, buf) != 0) return -1;
        for (int j = 0; j < 16; j++) {
            dir_entry_t* entry = (dir_entry_t*)&buf[j * 32];
            if (entry->name[0] == 0x00) return 0;
            if (entry->name[0] == 0xE5) continue;
            if (entry->attr == 0x0F) continue;
            if (entry->attr & 0x08) continue;
            fat_name_to_str(entry->name, info.name);
            info.attr = entry->attr;
            info.size = entry->file_size;
            info.first_cluster = entry->first_cluster;
            if (callback) callback(&info);
        }
    }
    return 0;
}

int fat_find(const char* name, file_info_t* info)
{
    unsigned int root_start = g_fs.reserved_sectors + g_fs.num_fats * g_fs.sectors_per_fat;
    unsigned char buf[512];
    unsigned char fat_name[8], fat_ext[3];
    str_to_fat_name(name, fat_name, fat_ext);

    for (unsigned int i = 0; i < g_fs.root_dir_sectors; i++) {
        if (disk_read(root_start + i, buf) != 0) return -1;
        for (int j = 0; j < 16; j++) {
            dir_entry_t* entry = (dir_entry_t*)&buf[j * 32];
            if (entry->name[0] == 0x00) return -1;
            if (entry->name[0] == 0xE5) continue;
            if (entry->attr == 0x0F) continue;
            if (entry->attr & 0x08) continue;
            if (str_casecmp((const char*)entry->name, (const char*)fat_name, 8) == 0 &&
                str_casecmp((const char*)&entry->name[8], (const char*)fat_ext, 3) == 0) {
                fat_name_to_str(entry->name, info->name);
                info->attr = entry->attr;
                info->size = entry->file_size;
                info->first_cluster = entry->first_cluster;
                return 0;
            }
        }
    }
    return -1;
}

file_handle_t* fat_open(const char* name)
{
    file_info_t info;
    if (fat_find(name, &info) != 0) return 0;
    if (info.attr & 0x10) return 0;
    str_copy(g_handle.name, info.name, 13);
    g_handle.size = info.size;
    g_handle.first_cluster = info.first_cluster;
    g_handle.current_cluster = info.first_cluster;
    g_handle.current_offset = 0;
    g_handle.cluster_offset = 0;
    g_handle.is_open = 1;
    return &g_handle;
}

int fat_read(file_handle_t* handle, unsigned char* buf, unsigned int size)
{
    if (!handle->is_open) return -1;
    unsigned int bytes_read = 0;
    unsigned int remaining = size;
    if (handle->current_offset >= handle->size) return 0;
    if (handle->current_offset + size > handle->size) remaining = handle->size - handle->current_offset;

    unsigned char sector_buf[512];

    while (remaining > 0) {
        if (handle->cluster_offset >= g_fs.sectors_per_cluster * 512) {
            handle->current_cluster = fat_next_cluster(handle->current_cluster);
            if (handle->current_cluster == 0xFFFF || handle->current_cluster == 0xFFF7) break;
            handle->cluster_offset = 0;
        }
        unsigned int sector = cluster_to_sector(handle->current_cluster) + (handle->cluster_offset / 512);
        unsigned int offset_in_sector = handle->cluster_offset % 512;
        if (disk_read(sector, sector_buf) != 0) break;
        unsigned int to_read = 512 - offset_in_sector;
        if (to_read > remaining) to_read = remaining;
        for (unsigned int i = 0; i < to_read; i++) buf[bytes_read + i] = sector_buf[offset_in_sector + i];
        bytes_read += to_read;
        remaining -= to_read;
        handle->current_offset += to_read;
        handle->cluster_offset += to_read;
    }
    return bytes_read;
}

void fat_close(file_handle_t* handle)
{
    if (handle) handle->is_open = 0;
}

void fat_debug(void)
{
}

#ifndef FAT_H
#define FAT_H

// FAT16 BPB (BIOS Parameter Block) - 位于引导扇区偏移 11 处
typedef struct {
    unsigned short bytes_per_sector;     // 偏移 11: 通常是 512
    unsigned char  sectors_per_cluster;  // 偏移 13
    unsigned short reserved_sectors;     // 偏移 14: 引导扇区数
    unsigned char  num_fats;             // 偏移 16: FAT 表数量 (通常是 2)
    unsigned short root_entries;         // 偏移 17: 根目录条目数
    unsigned short total_sectors_16;     // 偏移 19: 总扇区数 (< 65536)
    unsigned char  media_type;           // 偏移 21: 0xF8 = 硬盘
    unsigned short sectors_per_fat;      // 偏移 22: 每个 FAT 的扇区数
    unsigned short sectors_per_track;    // 偏移 24
    unsigned short num_heads;            // 偏移 26
    unsigned int   hidden_sectors;       // 偏移 28
    unsigned int   total_sectors_32;     // 偏移 32
} __attribute__((packed)) bpb_t;

// FAT16 根目录条目 - 32 字节
typedef struct {
    unsigned char  name[8];              // 文件名 (不足 8 字节用空格填充)
    unsigned char  ext[3];               // 扩展名
    unsigned char  attr;                 // 属性: 0x10=目录, 0x20=存档, 0x0F=长文件名
    unsigned char  reserved;             // 保留
    unsigned char  create_time_tenth;    // 创建时间(十分之一秒)
    unsigned short create_time;          // 创建时间
    unsigned short create_date;          // 创建日期
    unsigned short access_date;          // 最后访问日期
    unsigned short first_cluster_high;   // 起始簇号高 16 位 (FAT32 用)
    unsigned short write_time;           // 最后写入时间
    unsigned short write_date;           // 最后写入日期
    unsigned short first_cluster;        // 起始簇号
    unsigned int   file_size;            // 文件大小 (字节)
} __attribute__((packed)) dir_entry_t;

// FAT16 文件系统信息结构
typedef struct {
    unsigned char  sectors_per_cluster;
    unsigned short reserved_sectors;
    unsigned char  num_fats;
    unsigned short root_entries;
    unsigned short sectors_per_fat;
    unsigned int   root_dir_sectors;
    unsigned int   first_data_sector;
    unsigned int   total_clusters;
    unsigned char  fat_loaded;
    unsigned short fat_cache[512 * 128];  // 缓存整个 FAT 表 (最多 128 扇区)
} fat_fs_t;

// 文件信息
typedef struct {
    char           name[13];          // 8.3 格式文件名
    unsigned char  attr;
    unsigned int   size;
    unsigned short first_cluster;
} file_info_t;

// 文件句柄
typedef struct {
    char           name[13];
    unsigned int   size;
    unsigned short first_cluster;
    unsigned short current_cluster;
    unsigned int   current_offset;     // 文件内偏移
    unsigned int   cluster_offset;     // 当前簇内偏移
    unsigned char  is_open;
} file_handle_t;

// 初始化文件系统，成功返回 0，失败返回 -1
int fat_init(void);

// 列出根目录所有文件，返回文件数量
int fat_list_root(void (*callback)(const file_info_t*));

// 按文件名查找，返回 0 表示找到，-1 表示未找到
int fat_find(const char* name, file_info_t* info);

// 打开文件，返回句柄指针，失败返回 0
file_handle_t* fat_open(const char* name);

// 读取文件数据，返回实际读取的字节数
int fat_read(file_handle_t* handle, unsigned char* buf, unsigned int size);

// 关闭文件
void fat_close(file_handle_t* handle);

// 打印调试信息
void fat_debug(void);
