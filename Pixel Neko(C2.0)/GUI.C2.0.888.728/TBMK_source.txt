[boot.asm]
============================================================

org 0x7C00
bits 16

start:
    jmp short real_start
    nop

; ============ FAT16 BPB（BIOS需要）============
bpb_oem:            db "T.B.M.K. "
bpb_bytes_per_sec:  dw 512
bpb_sec_per_cluster: db 1
bpb_reserved:       dw 1
bpb_num_fats:       db 2
bpb_root_entries:   dw 224
bpb_total_sec16:    dw 2880
bpb_media:          db 0xF0
bpb_sec_per_fat:    dw 9
bpb_sec_per_track:  dw 18
bpb_num_heads:      dw 2
bpb_hidden:         dd 0
bpb_total_sec32:    dd 0
bpb_drive_num:      db 0x00
bpb_reserved2:      db 0
bpb_ext_sig:        db 0x29
bpb_vol_id:         dd 0x12345678
bpb_vol_label:      db "TBMK_GUI   "
bpb_fs_type:        db "FAT12   "

real_start:
    cli
    xor ax, ax
    mov ds, ax
    mov es, ax
    mov ss, ax
    mov sp, 0x7B00
    mov [boot_drive], dl

    in al, 0x92
    or al, 2
    out 0x92, al

    mov ah, 0x00
    mov dl, [boot_drive]
    int 0x13

    ; 读取内核
    mov ax, 0x0210
    mov cx, 0x0002
    mov dh, 0
    mov dl, [boot_drive]
    mov bx, 0x1000
    int 0x13
    jc disk_error

load_done:
    mov ax, 0x0013
    int 0x10

enter_protected:
    cli
    lgdt [gdt_ptr]
    mov eax, cr0
    or eax, 1
    mov cr0, eax
    jmp 0x08:0x1000

disk_error:
    mov si, err
    mov ah, 0x0E
.l2:
    lodsb
    test al, al
    jz $
    int 0x10
    jmp .l2

boot_drive db 0
err db 'E', 0

gdt_start:
    dd 0, 0
    dd 0x0000FFFF, 0x00CF9A00
    dd 0x0000FFFF, 0x00CF9200
gdt_end:
gdt_ptr:
    dw gdt_end - gdt_start - 1
    dd gdt_start

times 510 - ($ - $$) db 0
dw 0xAA55

static void draw_ok_button(unsigned char* v, int bx, int by, int pressed)
{
    for (int y = by; y < by+12; y++)
        for (int x = bx; x < bx+32; x++)
            v[y*320+x] = 7;
    if (pressed) {
        for (int x = bx; x < bx+32; x++) v[by*320+x] = 0;
        for (int y = by; y < by+12; y++) v[y*320+bx] = 0;
        for (int x = bx; x < bx+32; x++) v[(by+11)*320+x] = 15;
        for (int y = by; y < by+12; y++) v[y*320+bx+31] = 15;
        font_small_draw_text(v, bx+9, by+3, "OK", 0);
    } else {
        for (int x = bx; x < bx+32; x++) v[by*320+x] = 15;
        for (int y = by; y < by+12; y++) v[y*320+bx] = 15;
        for (int x = bx; x < bx+32; x++) v[(by+11)*320+x] = 0;
        for (int y = by; y < by+12; y++) v[y*320+bx+31] = 0;
        font_small_draw_text(v, bx+8, by+2, "OK", 0);
    }
}

static void draw_win(unsigned char* v, int wx, int wy)
{
    int ww = 160, wh = 112;
    for (int y = wy+1; y < wy+wh+1 && y < 200; y++)
        for (int x = wx+1; x < wx+ww+1 && x < 320; x++)
            v[y*320+x] = 0;
    for (int y = wy; y < wy+wh && y < 200; y++)
        for (int x = wx; x < wx+ww && x < 320; x++)
            v[y*320+x] = (y==wy||y==wy+wh-1||x==wx||x==wx+ww-1)?0:(y<wy+12)?1:15;
    for (int y = wy+2; y < wy+10; y++)
        for (int x = wx+ww-14; x < wx+ww-2; x++)
            v[y*320+x] = (y==wy+2||x==wx+ww-14)?15:(y==wy+9||x==wx+ww-3)?0:7;
    font_small_draw_text(v, wx+5, wy+3, "About", 15);

    int lx = wx + (ww - 52) / 2;
    int ly = wy + 14;
    for (int y = ly+3; y < ly+18; y++)
        for (int x = lx-7; x < lx+48; x++) v[y*320+x] = 1;
    for (int x = lx-4; x < lx+45; x++) v[(ly+3)*320+x] = 9;
    for (int y = ly; y < ly+12; y++)
        for (int x = lx; x < lx+12; x++)
            v[y*320+x] = (y==ly||x==lx)?12:4;
    for (int y = ly; y < ly+12; y++)
        for (int x = lx+10; x < lx+22; x++)
            v[y*320+x] = (y==ly||x==lx+10)?10:2;
    for (int y = ly+7; y < ly+19; y++)
        for (int x = lx+19; x < lx+31; x++)
            v[y*320+x] = (y==ly+7||x==lx+19)?11:3;
    for (int y = ly+7; y < ly+19; y++)
        for (int x = lx+29; x < lx+41; x++)
            v[y*320+x] = (y==ly+7||x==lx+29)?13:5;
    font_small_draw_char(v, lx+4, ly+4, 'T', 15);
    font_small_draw_char(v, lx+14, ly+4, 'B', 15);
    font_small_draw_char(v, lx+23, ly+11, 'M', 15);
    font_small_draw_char(v, lx+33, ly+11, 'K', 15);

    int line_y = wy + 40;
    for (int x = wx+4; x < wx+ww-4; x++) v[line_y*320+x] = 8;

    font_small_draw_text(v, wx+8, wy+46, VER_STRING1, 0);
    font_small_draw_text(v, wx+8, wy+54, VER_STRING2, 0);
    font_small_draw_text(v, wx+8, wy+64, VER_STRING3, 8);

    int bx = wx + ww - 42, by = wy + 90;
    draw_ok_button(v, bx, by, 0);
}

static void draw_box(unsigned char* v, int x, int y, int w, int h)
{
    for (int i = x; i < x+w; i+=2) {
        if (i < 320 && y < 200) v[y*320+i] = 0;
        if (i < 320 && y+h-1 < 200) v[(y+h-1)*320+i] = 0;
    }
    for (int i = y; i < y+h; i+=2) {
        if (x < 320 && i < 200) v[i*320+x] = 0;
        if (x+w-1 < 320 && i < 200) v[i*320+x+w-1] = 0;
    }
}

static void draw_icon(unsigned char* v, int x, int y)
{
    for (int i = y; i < y+16 && i < 200; i++)
        for (int j = x; j < x+16 && j < 320; j++)
            v[i*320+j] = 1;
    for (int j = x; j < x+16; j++) v[y*320+j] = 9;
    for (int i = y; i < y+16; i++) v[i*320+x] = 9;
    for (int j = x; j < x+16; j++) v[(y+15)*320+j] = 0;
    for (int i = y; i < y+16; i++) v[i*320+x+15] = 0;
    for (int i = y+2; i < y+12; i++)
        for (int j = x+6; j < x+10; j++)
            v[i*320+j] = 15;
    for (int i = y+12; i < y+14; i++)
        for (int j = x+6; j < x+10; j++)
            v[i*320+j] = 15;
    font_small_draw_text(v, x-2, y+18, "About", 0);
}

static void animate_close(unsigned char* v, int wx, int wy)
{
    int ww = 160, wh = 100;
    int cx = wx + ww/2;
    int cy = wy + wh/2;
    for (int step = 10; step >= 0; step--) {
        int nw = ww * step / 10;
        int nh = wh * step / 10;
        if (nw < 4) nw = 4;
        if (nh < 4) nh = 4;
        int nx = cx - nw/2;
        int ny = cy - nh/2;
        for (int i = 0; i < 320*200; i++) v[i] = 7;
        draw_icon(v, 20, 156);
        for (int yi = ny; yi < ny+nh && yi < 200; yi++)
            for (int xi = nx; xi < nx+nw && xi < 320; xi++)
                v[yi*320+xi] = (yi==ny||yi==ny+nh-1||xi==nx||xi==nx+nw-1)?0:(yi<ny+12)?1:15;
        for (volatile int i = 0; i < 20000; i++);
    }
}

void kernel_main(void)
{
    unsigned char* v = (unsigned char*)0xA0000;
    int mx, my, mb;
    int last_mx = -1, last_my = -1;

    int wx = 80, wy = 44, ww = 160, wh = 100;
    int win_visible = 1;
    int dragging = 0, drag_ox = 0, drag_oy = 0;
    int old_wx = wx, old_wy = wy;
    int last_mb = 0;

    for (int i = 0; i < 320*200; i++) v[i] = 7;
    draw_icon(v, 20, 156);
    draw_win(v, wx, wy);
    mouse_init();

    while (1) {
        if (inb(0x64) & 0x01) {
            unsigned char status = inb(0x64);
            if (status & 0x20) mouse_handle_byte(inb(0x60));
            else inb(0x60);
        }
        mouse_get(&mx, &my, &mb);
        int mb_press = (mb & 1) && !(last_mb & 1);
        last_mb = mb;

        if (mb_press && !win_visible) {
            if (mx >= 20 && mx < 36 && my >= 156 && my < 172) {
                win_visible = 1;
                for (int i = 0; i < 320*200; i++) v[i] = 7;
                draw_icon(v, 20, 156);
                draw_win(v, wx, wy);
                mouse_reset(); last_mx = -1; last_my = -1;
            }
        }
        if (mb_press && win_visible) {
            int bx = wx + ww - 42, by = wy + 90;
            if (mx >= bx && mx < bx+32 && my >= by && my < by+12) {
                draw_ok_button(v, bx, by, 1);
                for (volatile int i = 0; i < 30000; i++);
                win_visible = 0;
                animate_close(v, wx, wy);
                for (int i = 0; i < 320*200; i++) v[i] = 7;
                draw_icon(v, 20, 156);
                mouse_reset(); last_mx = -1; last_my = -1;
            }
        }
        if (mb & 1 && win_visible && !dragging) {
            if (mx >= wx+ww-14 && mx < wx+ww-2 && my >= wy+2 && my < wy+10) {
                win_visible = 0;
                animate_close(v, wx, wy);
                for (int i = 0; i < 320*200; i++) v[i] = 7;
                draw_icon(v, 20, 156);
                mouse_reset(); last_mx = -1; last_my = -1;
                mouse_draw(v); last_mx = mx; last_my = my;
                while (mb & 1) {
                    if (inb(0x64) & 0x01) {
                        unsigned char st = inb(0x64);
                        if (st & 0x20) mouse_handle_byte(inb(0x60));
                    }
                    mouse_get(&mx, &my, &mb);
                }
                continue;
            }
        }
        if (win_visible && mb & 1) {
            if (!dragging) {
                if (mx >= wx && mx < wx+ww && my >= wy && my < wy+12) {
                    dragging = 1; drag_ox = mx-wx; drag_oy = my-wy;
                    old_wx = wx; old_wy = wy;
                }
            }
            if (dragging) {
                int nwx = mx-drag_ox, nwy = my-drag_oy;
                if (nwx < 0) nwx = 0; if (nwy < 0) nwy = 0;
                if (nwx+ww > 320) nwx = 320-ww;
                if (nwy+wh > 200) nwy = 200-wh;
                if (nwx != wx || nwy != wy) {
                    for (int i = 0; i < 320*200; i++) v[i] = 7;
                    draw_icon(v, 20, 156);
                    draw_box(v, nwx, nwy, ww, wh);
                    wx = nwx; wy = nwy;
                }
            }
        } else {
            if (dragging) {
                dragging = 0;
                for (int i = 0; i < 320*200; i++) v[i] = 7;
                draw_icon(v, 20, 156);
                draw_win(v, wx, wy);
                mouse_reset(); last_mx = -1; last_my = -1;
            }
        }
        if (!dragging && (mx != last_mx || my != last_my)) {
            mouse_draw(v);
            last_mx = mx; last_my = my;
        }
    }
}

#ifndef PORT_H
#define PORT_H

static inline void outb(unsigned short port, unsigned char value)
{
    __asm__ volatile("outb %0, %1" : : "a"(value), "Nd"(port));
}

static inline unsigned char inb(unsigned short port)
{
    unsigned char value;
    __asm__ volatile("inb %1, %0" : "=a"(value) : "Nd"(port));
    return value;
}

void font_draw_char(unsigned char* vga, int x, int y, char c, unsigned char fg)
{
    unsigned char* glyph = font[(unsigned char)c];
    for (int row = 0; row < 8; row++) {
        unsigned char line = glyph[row];
        for (int col = 0; col < 8; col++) {
            if (line & (1 << (7 - col))) {
                int px = x + col, py = y + row;
                if (px >= 0 && px < 320 && py >= 0 && py < 200)
                    vga[py*320+px] = fg;
            }
        }
    }
}

void font_draw_text(unsigned char* vga, int x, int y, const char* text, unsigned char fg)
{
    while (*text) {
        font_draw_char(vga, x, y, *text, fg);
        x += 8;
        text++;
    }
}

#include "port.h"

static int mx = 160, my = 100;
static int btn = 0;
static int cycle = 0;
static unsigned char buf[3];
static int init_done = 0;

static unsigned char saved[9][9];
static int saved_x = -1, saved_y = -1;

static void wait_kbc(void)
{
    for (volatile int i = 0; i < 10000; i++) {
        if (!(inb(0x64) & 0x02)) return;
    }
}

static void wait_read(void)
{
    for (volatile int i = 0; i < 10000; i++) {
        if (inb(0x64) & 0x01) return;
    }
}

void mouse_init(void)
{
    while (inb(0x64) & 0x01) inb(0x60);
    wait_kbc(); outb(0x64, 0xA8);
    wait_kbc(); outb(0x64, 0x20);
    wait_read();
    unsigned char cfg = inb(0x60);
    cfg |= 0x02;
    wait_kbc(); outb(0x64, 0x60);
    wait_kbc(); outb(0x60, cfg);
    wait_kbc(); outb(0x64, 0xD4);
    wait_kbc(); outb(0x60, 0xF4);
    wait_read();
    if (inb(0x60) == 0xFA) init_done = 1;
    wait_kbc(); outb(0x64, 0xD4); wait_kbc(); outb(0x60, 0xF3);
    wait_read(); inb(0x60);
    wait_kbc(); outb(0x64, 0xD4); wait_kbc(); outb(0x60, 60);
    wait_read(); inb(0x60);
}

void mouse_handle_byte(unsigned char d)
{
    if (!init_done) return;
    if (cycle == 0 && !(d & 0x08)) return;
    buf[cycle] = d;
    cycle++;
    if (cycle == 3) {
        cycle = 0;
        btn = buf[0] & 0x07;
        int dx = buf[1];
        int dy = buf[2];
        if (buf[0] & 0x10) dx = dx - 256;
        if (buf[0] & 0x20) dy = dy - 256;
        mx += dx / 2;
        my -= dy / 2;
        if (mx < 0) mx = 0;
        if (mx > 319) mx = 319;
        if (my < 0) my = 0;
        if (my > 199) my = 199;
    }
}

void mouse_get(int* x, int* y, int* b)
{
    if (x) *x = mx;
    if (y) *y = my;
    if (b) *b = btn;
}

void mouse_draw(unsigned char* vga)
{
    if (!init_done) return;
    if (saved_x >= 0) {
        for (int i = 0; i < 9; i++)
            for (int j = 0; j < 9; j++)
                if (saved_y+i < 200 && saved_x+j < 320)
                    vga[(saved_y+i)*320+(saved_x+j)] = saved[i][j];
    }
    int x = mx, y = my;
    for (int i = 0; i < 9; i++)
        for (int j = 0; j < 9; j++)
            saved[i][j] = (y+i < 200 && x+j < 320) ? vga[(y+i)*320+(x+j)] : 0;
    saved_x = x;
    saved_y = y;
    for (int j = 0; j < 9; j++) {
        int px = x+j, py = y+4;
        if (px < 320 && py < 200) vga[py*320+px] = 0;
    }
    for (int i = 0; i < 9; i++) {
        int px = x+4, py = y+i;
        if (px < 320 && py < 200) vga[py*320+px] = 0;
    }
    if (x+4 < 320 && y+4 < 200) vga[(y+4)*320+(x+4)] = 15;
}

void mouse_reset(void)
{
    saved_x = -1;
    saved_y = -1;
}

@echo off

title TBMK GUI 编译

del *.o *.bin *.elf os.img pad.bin 2>nul

nasm -f bin boot.asm -o boot.bin || (pause & exit /b)
nasm -f elf32 kernel_entry.asm -o entry.o || (pause & exit /b)
gcc -m32 -ffreestanding -nostdlib -nostartfiles -fno-builtin -Os -c kernel.c -o kernel.o || (pause & exit /b)
gcc -m32 -ffreestanding -nostdlib -nostartfiles -fno-builtin -Os -c mouse.c -o mouse.o || (pause & exit /b)
gcc -m32 -ffreestanding -nostdlib -nostartfiles -fno-builtin -Os -c font.c -o font.o || (pause & exit /b)
gcc -m32 -ffreestanding -nostdlib -nostartfiles -fno-builtin -Os -c font_small.c -o font_small.o || (pause & exit /b)
gcc -m32 -ffreestanding -nostdlib -nostartfiles -T link.ld entry.o kernel.o mouse.o font.o font_small.o -o kernel.elf || (pause & exit /b)
objcopy -O binary kernel.elf kernel.bin
copy /b boot.bin + kernel.bin os.img > nul

for %%A in (kernel.bin) do echo 内核: %%~zA 字节
del *.o *.elf 2>nul
